1 - Variables and Data Types

let studentName = "Zhangirkhan";
let age = 20;
let isActive = true;
let courses = ["Math", "Physics", "React"];
let address = { city: "Almaty", street: "Askarov" };

let middleName;      // undefined, didn't set it

console.log(typeof studentName, typeof age, typeof isActive);
console.log(typeof courses); // object, even though it's an array
console.log(`${studentName} is ${age} years old and lives in ${address.city}`);

let vs const - const just means you can't reassign the variable itself. if it's an object you can still change its fields, so const doesn't really mean "frozen".

typeof null returns "object", which is wrong/weird and everyone agrees it's a bug that's just stuck in the language forever.

primitives: string, number, boolean, undefined, null, symbol, bigint. arrays and objects are not primitives, they're references.

name/age/isActive get copied by value, courses/address get copied by reference (see task 5, that's where it actually matters).

2 - Arrays

const numbers = [3, 7, 2, 10, 5];

const doubled = numbers.map(n => n * 2);
const above5 = numbers.filter(n => n > 5);
const firstAbove5 = numbers.find(n => n > 5);
let total = numbers.reduce((sum, n) => sum + n, 0);
const has10 = numbers.includes(10);

console.log(numbers); // still [3,7,2,10,5], untouched

map/filter/reduce don't mutate the array, they just return new stuff, which is why the original numbers array is still the same at the end.

5 - Values and References

let original = { name: "Alice", score: 10 };
let copy = original;
copy.score = 99;
console.log(original.score); // 99 too! same object in memory

let original2 = { name: "Alice", score: 10 };
let realCopy = { ...original2 };
realCopy.score = 50;
console.log(original2.score); // still 10, spread made a new object

let user2 = { name: "Alice", address: { city: "Almaty" } };
let shallow = { ...user2 };
shallow.address.city = "Astana";
console.log(user2.address.city); // Astana!! spread is shallow, address itself wasn't copied

let deep = { ...user2, address: { ...user2.address } };
deep.address.city = "Karaganda";
console.log(user2.address.city); // still Astana from before, unaffected now

copy = original doesn't copy anything, both variables just point at the same object.

spread only copies one level deep. the address object inside is still shared unless you spread it too - this one got me for a second honestly.

7 - Functions as Values

function add(a, b) { return a + b; }
function multiply(a, b) { return a * b; }

function calculate(a, b, operation) {
  return operation(a, b);
}

console.log(calculate(5, 3, add));
console.log(calculate(5, 3, multiply));

yes, functions can be stored in variables and passed around like any other value (that's what's happening with "operation" above).

add by itself is just a reference to the function. add() actually calls it and gives you back a number.

8 - Scope

const message = "global";

function scopeTest() {
  let message = "function";
  if (true) {
    let message = "block";
    console.log(message); // block
  }
  console.log(message); // function, if-block doesn't leak
}
scopeTest();
console.log(message); // global

{
  var v = "var value";
  let l = "let value";
  const c = "const value";
}
console.log(v); // works, var ignores block scope
// console.log(l) here would just throw, not defined outside the {}

global scope = whole file can see it. function scope = only inside that function. block scope = only inside the {} it was declared in.

var doesn't care about blocks (if/for/{}), only functions - that's basically the main reason people avoid it now and just use let/const.

9 - Closure

function createCounter() {
  let count = 0;
  return function () {
    count = count + 1;
    return count;
  };
}

const counter1 = createCounter();
const counter2 = createCounter();
console.log(counter1()); // 1
console.log(counter1()); // 2
console.log(counter1()); // 3
console.log(counter2()); // 1, separate counter

function createAdder(value) {
  return num => num + value;
}
const addFive = createAdder(5);
console.log(addFive(10)); // 15
console.log(addFive(20)); // 25

the inner function keeps a link to the variables around it even after createCounter is done running - that's the closure. each call to createCounter makes a fresh "count" so counter1 and counter2 don't share anything.

10 - Destructuring, Spread and Rest

const numbers = [10, 20, 30, 40];
const [first, second] = numbers;

const user = { id: 1, name: "Aigerim", age: 21 };
const { name, age } = user;

const moreNumbers = [...numbers, 50];
const olderUser = { ...user, age: 22 };
const userWithEmail = { ...user, email: "aigerim@mail.com" };
const combined = [1, 2].concat ? [...[1, 2], ...[3, 4]] : null; // just showing spread works for arrays too

function sum(...nums) {
  let total = 0;
  for (const n of nums) total += n;
  return total;
}
console.log(sum(1, 2));
console.log(sum(1, 2, 3, 4));

spread = unpacking values out of an array/object. rest = the opposite, packing a bunch of loose arguments into one array. same three dots, different job depending on which side it's on.

11 - Optional Chaining and Default Values

const userA = { name: "Zhangirkhan", address: { city: "Almaty" } };
const userB = { name: "Timur" }; // no address at all

// userB.address.city would just crash the page here
console.log(userA.address?.city);
console.log(userB.address?.city); // undefined, no crash

const cityA = userA.address?.city ?? "City not specified";
const cityB = userB.address?.city ?? "City not specified";

const values = [0, "", false, null, undefined];
console.log(values.map(v => v || "fallback"));
console.log(values.map(v => v ?? "fallback"));

?. just stops and gives back undefined instead of erroring when something in the chain is missing.

?? only falls back on null/undefined. || falls back on basically anything "falsy" - 0, empty string, false too - which is annoying if 0 or "" was actually a real value you wanted to keep. this is the difference people usually get wrong.

const students = [
  { id: 1, name: "Aigerim", age: 20, grades: [85, 90, 78] },
  { id: 2, name: "Nurlan", age: 21, grades: [60, 55, 70] },
  { id: 3, name: "Dana", age: 19, grades: [91, 95, 88] },
  { id: 4, name: "Yerlan", age: 22, grades: [50, 65, 40] },
  { id: 5, name: "Aidos", age: 20, grades: [72, 68, 80] },
];

function getAverage(grades) {
  let sum = 0;
  for (const g of grades) sum += g;
  return sum / grades.length;
}

const getStudentAverage = student => getAverage(student.grades);

function getPassedStudents(students) {
  return students.filter(s => getStudentAverage(s) >= 60);
}

const getStudentNames = students => students.map(s => s.name);

function findStudent(students, id) {
  return students.find(s => s.id === id);
}

function getTopStudent(students) {
  let top = students[0];
  for (const s of students) {
    if (getStudentAverage(s) > getStudentAverage(top)) top = s;
  }
  return top;
}

const summary = students.map(s => ({
  id: s.id,
  name: s.name,
  average: Math.round(getStudentAverage(s) * 100) / 100,
  passed: getStudentAverage(s) >= 60,
}));
